To create an arc shape with `Konva`, we can instantiate a `Konva.Arc()` object.

For full list of properties and methods, see the [Arc API Reference](/api/Konva.Arc.html).

import Tabs from "@theme/Tabs";

  <TabItem value="Vanilla" default>
```js live vanilla
import Konva from 'konva';

const stage = new Konva.Stage({
  container: 'container',
  width: window.innerWidth,
  height: window.innerHeight
});

const layer = new Konva.Layer();
stage.add(layer);

const arc = new Konva.Arc({
  x: stage.width() / 2,
  y: stage.height() / 2,
  innerRadius: 40,
  outerRadius: 70,
  angle: 60,
  fill: 'yellow',
  stroke: 'black',
  strokeWidth: 4
});

layer.add(arc);
```

import { Stage, Layer, Arc } from 'react-konva';

const App = () => {
  return (
    <Stage width={window.innerWidth} height={window.innerHeight}>
      <Layer>

          x={window.innerWidth / 2}
          y={window.innerHeight / 2}
          innerRadius={40}
          outerRadius={70}
          angle={60}
          fill="yellow"
          stroke="black"
          strokeWidth={4}
        />
      </Layer>
    </Stage>
  );
};

export default App;

import { Component } from '@angular/core';

      </ko-stage>

    height: window.innerHeight,
  };
  public configArc: ArcConfig = {

    y: window.innerHeight / 2,
